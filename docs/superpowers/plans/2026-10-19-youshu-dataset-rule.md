```js
function buildRule(meta, script) {
  const rule = { domain: meta.domain };
  if (meta.url_pattern !== undefined) rule.url_pattern = meta.url_pattern;
  rule.parser_type = meta.parser_type;
  if (meta.dataset !== undefined) rule.dataset = meta.dataset;
  if (meta.selectors !== undefined) rule.selectors = meta.selectors;
  if (script !== null) rule.script = script;
  rule.config = meta.config;
  rule.version = meta.version;
  rule.author = meta.author;
  rule.language = meta.language;
  rule.description = meta.description;
  return rule;
}

// A parser.js / dataset.js file must be one object-literal expression.
function evalScript(folder, file, text) {
  let obj;
  try { obj = new Function('return ' + text)(); }
  catch (e) { throw new Error(`${folder}/${file}: not a valid expression — ${e.message}`); }
  if (typeof obj !== 'object' || obj === null) throw new Error(`${folder}/${file}: must evaluate to an object ({ ... })`);
  return obj;
}
```

```js
// Node twin of the ctx the extension builds for dataset rules
// (NovelPacker-Extension src/apps/sandbox/sandbox.ts: html '', url =
// dataset.origin, plus params and task) and of the checks DatasetRunner's
// normalizeResult applies to run() results. Relative fetch URLs resolve against
// the origin, as the engine's in-tab fetch does. Not covered by the drift check:
// keep it in step with those two files by hand.
import { createScriptableContext } from './scriptable-context.mjs';

const STATUSES = new Set(['ok', 'notfound', 'retry', 'wall']);
const TABLE = /^\w[\w.-]*$/;

export function loadDatasetScript(text) {
  const obj = new Function('return ' + text.trim())();
  if (!obj || typeof obj.seed !== 'function' || typeof obj.run !== 'function') {
    throw new Error('dataset script must define seed and run');
  }
  return obj;
}

export function createDatasetContext({ origin, config = {}, params = {}, task = null, fetch }) {
  const fromOrigin = (url, init) => fetch(new URL(url, origin).toString(), init);
  return { ...createScriptableContext({ html: '', url: origin, config, fetch: fromOrigin }), params, task };
}

export function checkRunResult(result) {
  if (!result || !STATUSES.has(result.status)) throw new Error(`Invalid run status: ${JSON.stringify(result?.status)}`);
  const records = result.records ?? [];
  const tasks = result.tasks ?? [];
  if (!Array.isArray(records) || !Array.isArray(tasks)) throw new Error('run: records and tasks must be arrays');
  for (const r of records) {
    if (!r || typeof r.key !== 'string' || !r.key || typeof r.table !== 'string' || !TABLE.test(r.table)) {
      throw new Error(`Invalid record: ${JSON.stringify(r)?.slice(0, 200)}`);
    }
  }
  for (const t of tasks) {
    if (!t || typeof t.key !== 'string' || !t.key) throw new Error(`Invalid task: ${JSON.stringify(t)?.slice(0, 200)}`);
  }
  if (result.status === 'wall' && !result.message) throw new Error('wall needs a message');
  return result;
}

export async function runTask(script, { origin, config, params, task, fetch }) {
  const rule = loadDatasetScript(script);
  const ctx = createDatasetContext({ origin, config, params, task, fetch });
  return checkRunResult(await rule.run(ctx, task));
}
```

```js
({
  // One task per book ID; review pages are discovered from each book page.
  async seed(ctx) {
    const p = ctx.params || {};
    const from = Number(p.from ?? 1);
    const to = Number(p.to ?? 330000);
    if (!Number.isInteger(from) || !Number.isInteger(to) || from < 1 || to < from) {
      throw new Error(`Invalid book ID range: from=${p.from} to=${p.to}`);
    }
    const tasks = [];
    for (let id = from; id <= to; id++) tasks.push({ key: `book:${id}`, kind: 'book', id });
    return tasks;
  },

  async run(ctx, task) {
    const p = ctx.params || {};
    const opts = { mode: p.mode === 'full' ? 'full' : 'update', since: String(p.since ?? ''), reviews: p.reviews !== false };
    if (task.kind === 'book') return this.book(ctx, task, opts);
    if (task.kind === 'reviews') return this.reviews(ctx, task, opts);
    throw new Error(`Unknown task kind: ${task.kind}`);
  },

  async book(ctx, task, opts) {
    const d = this.doc(await ctx.fetch(`/book/${task.id}`));
    const titleEl = d.querySelector('.divbox.cf.blockn span[style*="font-size:20px"]');
    if (!titleEl) {
      const box = this.text(d.querySelector('#content .msgbox'));
      if (box.includes('不存在')) return { status: 'notfound' };
      return { status: 'retry', message: box || `No book block on /book/${task.id}: ${this.text(d.querySelector('title'))}` };
    }

    const info = {};
    for (const td of d.querySelectorAll('.tabcontent .tabvalue table.hide td')) {
      const t = this.text(td);
      const i = t.indexOf('：');
      if (i > 0) info[t.slice(0, i).replace(/\s+/g, '')] = t.slice(i + 1).trim();
    }

    const histogramArchived = {};
    for (const bar of d.querySelectorAll('#project-desc .book-score-charts .bar-item')) {
      histogramArchived[this.text(bar.querySelector('.bar-name')).replace(/\s+/g, '')] = this.text(bar.querySelector('.bar-rate')).replace(/\s+/g, '');
    }
    const histogramLive = {};
    for (const row of d.querySelectorAll('.star-rating-chart tr')) {
      const m = this.text(row).match(/(\d)\s*星\s*([\d.]+\s*%)/);
      if (m) histogramLive[`${m[1]}星`] = m[2].replace(/\s+/g, '');
    }

    // Newest date among the book page's 最新书评 previews.
    const times = [...d.querySelectorAll('.comment-box .c_tag .c_label')]
      .filter((l) => l.textContent.includes('时间'))
      .map((l) => this.text(l.nextElementSibling))
      .filter(Boolean)
      .sort();
    const latestReviewAt = times.length ? times[times.length - 1] : null;
    const live = d.querySelector('.ratenum');
    const overview = d.querySelector('#project-desc .book-score-overview');

    const book = {
      id: task.id,
      title: this.text(titleEl),
      author: this.text(titleEl.nextElementSibling?.querySelector('a')),
      status: info['连载状态'] ?? null,
      tags: [...d.querySelectorAll('.divbox.cf.blockn a.tag-link')].map((a) => this.text(a)),
      info,
      wordCount: this.num(info['全文字数']),
      updatedAt: info['最后更新'] || null,
      cover: this.abs(d.querySelector('.book-detail-img img')?.getAttribute('src')),
      intro: this.multiline(d.querySelector('.tabcontent .tabvalue:first-child')),
      scoreArchived: this.num(this.text(overview?.querySelector('.score'))),
      scoreArchivedCount: this.num(this.text(overview?.querySelector('.scorer-count'))),
      scoreLive: this.num(this.text(live)),
      scoreLiveCount: this.num(this.text(live?.nextElementSibling)),
      histogramArchived: Object.keys(histogramArchived).length ? histogramArchived : null,
      histogramLive: Object.keys(histogramLive).length ? histogramLive : null,
      latestReviewAt,
    };

    const tasks = [];
    if (opts.reviews && latestReviewAt && (opts.mode === 'full' || latestReviewAt >= opts.since)) {
      tasks.push({ key: `reviews:${task.id}:1`, kind: 'reviews', bookId: task.id, page: 1 });
    }
    return { status: 'ok', records: [{ table: 'books', key: String(task.id), data: book }], tasks };
  },

  async reviews(ctx, task, opts) {
    const d = this.doc(await ctx.fetch(`/reviews/${task.bookId}/${task.page}.html`));
    if ([...d.querySelectorAll('legend')].some((l) => l.textContent.includes('用户登录'))) {
      return { status: 'wall', message: 'Login required: open the crawl tab, log in to youshu.me, then Resume.' };
    }
    const cards = [...d.querySelectorAll('#jieqi_page_contents .pb.c_row.cf')];
    if (!cards.length) return { status: 'retry', message: `No reviews on ${task.key}: ${this.text(d.querySelector('title'))}` };

    const records = [];
    let oldest = null;
    for (const card of cards) {
      const rid = Number(card.querySelector('.c_subject a.pop')?.getAttribute('href')?.match(/\/reviewshow\/(\d+)\//)?.[1]);
      const userLink = card.querySelector('a[href*="userpage.php?id="]');
      const userId = Number(userLink?.getAttribute('href')?.match(/[?&]id=(\d+)/)?.[1]);
      if (!rid || !userId) throw new Error(`Unparsable review card on ${task.key}`);
      const meta = {};
      for (const label of card.querySelectorAll('.c_tag .c_label')) {
        meta[this.text(label).replace(/[：:]$/, '')] = this.text(label.nextElementSibling);
      }
      const content = this.multiline(card.querySelector('.c_description'));
      const userName = this.text(userLink.querySelector('p'));
      const review = {
        rid,
        bookId: task.bookId,
        userId,
        userName,
        title: this.text(card.querySelector('.c_subject a.pop')),
        content,
        deleted: content === '本评论已被清空。',
        stars: this.num(card.querySelector('.c_subject .rt [title$="颗星"]')?.getAttribute('title')),
        createdAt: meta['时间'] || null,
        likes: this.num(this.text(card.querySelector('[class^="love-count-"]'))) ?? 0,
        dislikes: this.num(this.text(card.querySelector('[class^="hate-count-"]'))) ?? 0,
        replies: this.num(meta['回复']) ?? 0,
        views: this.num(meta['点击']),
      };
      if (!review.createdAt) throw new Error(`Review ${rid} has no time on ${task.key}`);
      if (oldest === null || review.createdAt < oldest) oldest = review.createdAt;
      records.push({ table: 'reviews', key: String(rid), data: review });
      records.push({
        table: 'users',
        key: String(userId),
        data: {
          id: userId,
          name: userName,
          level: this.text(card.querySelector('.user-level')) || null,
          joinedAt: this.text(card.querySelector('.pop0')).replace('加入', '') || null,
        },
      });
    }

    const total = Number(this.text(d.querySelector('.pages #pagestats')).split('/')[1]) || 1;
    const next = (page) => ({ key: `reviews:${task.bookId}:${page}`, kind: 'reviews', bookId: task.bookId, page });
    const tasks = [];
    if (opts.mode === 'full') {
      if (task.page === 1) for (let page = 2; page <= total; page++) tasks.push(next(page));
    } else if (task.page < total && oldest >= opts.since) {
      tasks.push(next(task.page + 1));
    }
    return { status: 'ok', records, tasks };
  },

  doc(html) {
    return new DOMParser().parseFromString(html, 'text/html');
  },

  text(el) {
    return el ? (el.textContent || '').replace(/\s+/g, ' ').trim() : '';
  },

  // Text with <br> kept as newlines.
  multiline(el) {
    if (!el) return '';
    const html = el.innerHTML.replace(/<br\s*\/?>/gi, '\n');
    const tmp = new DOMParser().parseFromString(`<div>${html}</div>`, 'text/html').querySelector('div');
    return (tmp.textContent || '').split('\n').map((l) => l.trim()).join('\n').trim();
  },

  num(s) {
    const m = String(s ?? '').replace(/,/g, '').match(/-?\d+(?:\.\d+)?/);
    return m ? Number(m[0]) : null;
  },

  abs(url) {
    return url ? new URL(url, 'https://www.youshu.me/').toString() : null;
  },
})
```

In `docs/RULE_SYSTEM.md`, insert before the `## See also` heading:

````markdown
## 11. Dataset rules (`parser_type: "dataset"`)

A dataset rule does not produce a novel. It enumerates **tasks** and turns each
into **records**; the extension's Datasets view runs it through the crawl tab
(in-tab `fetch`, so a Cloudflare-fronted site works once that tab has passed the
challenge), keeps everything in IndexedDB with resume, and exports NDJSON.

Folder layout: `src/<site>-dataset/` with `rule.json`, `dataset.js`,
`samples.json`, `rule.test.mjs`, `fixtures/`. The domain is the rule's identity,
so use `<site>-dataset` — it never replaces the site's novel rule.

```json
{
  "domain": "example.com-dataset",
  "parser_type": "dataset",
  "dataset": { "origin": "https://www.example.com", "defaultParams": { "from": 1, "to": 100 } },
  "config": { "rate_limit_ms": 1000 },
  "version": "1.0.0", "author": "…", "language": "en", "description": "…"
}
```

No `selectors`. `dataset.js` is one object literal:

```js
({
  async seed(ctx) { return [{ key: 'item:1', id: 1 }]; },   // first tasks
  async run(ctx, task) {                                     // one task
    const html = await ctx.fetch('/item/' + task.id);        // relative → dataset.origin
    return { status: 'ok', records: [{ table: 'items', key: String(task.id), data: { … } }], tasks: [] };
  },
})
```

- `ctx` = the scriptable ctx (`fetch`, `utils`, `config`, `$`, DOM/DOMParser)
  with `html: ''`, `url: dataset.origin`, `params` (the job's params, defaults
  from `dataset.defaultParams`) and `task`.
- A task is a JSON object with a string `key`, unique within the job; child
  tasks are skipped when their key already exists.
- `run` returns `{ status, records?, tasks?, message? }`:
  - `ok` — upsert `records` by `(table, key)` (table matches `/^\w[\w.-]*$/`,
    key is a non-empty string) and enqueue `tasks`;
  - `notfound` — the item does not exist;
  - `retry` (or a throw) — tried 3 times, then `failed` (Retry failed re-queues);
  - `wall` — login/anti-bot page; needs a `message`, pauses the job and brings
    the crawl window forward. A Cloudflare page on any `ctx.fetch` pauses the
    job even if the script catches the error.
- Never return `ok` with empty records for a page that should have data — throw
  or `retry`.

Tooling: `npm run build` checks that `dataset.js` defines `seed` and `run` and
that `dataset.origin` is http(s); `src/_ctx/dataset-context.mjs` gives tests the
same ctx (`createDatasetContext`, `runTask`) and the engine's result checks
(`checkRunResult`); `npm run test:live <domain>` runs `seed(samples.params)` and
`run(samples.task)`. Sites behind Cloudflare cannot be fetched from Node: save
fixtures from a browser tab that passed the challenge and say so in
`fixtures/README.md` (example: `src/youshu.me-dataset/`).
````
